cmake_minimum_required(VERSION 3.20)
project(lingji_installer_ui LANGUAGES CXX)
add_library(lingji-ui SHARED theme.cpp)
if(MSVC)
  target_sources(lingji-ui PRIVATE exports.def)
else()
  target_sources(lingji-ui PRIVATE exports-mingw.def)
endif()
target_compile_features(lingji-ui PRIVATE cxx_std_17)
target_compile_definitions(lingji-ui PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A000002)
set_property(TARGET lingji-ui PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  target_compile_options(lingji-ui PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(lingji-ui PRIVATE -Wall -Wextra -Werror -pthread)
  target_link_options(lingji-ui PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--kill-at -pthread)
  set_target_properties(lingji-ui PROPERTIES PREFIX "")
endif()
target_link_libraries(lingji-ui PRIVATE user32 gdi32 comctl32 bcrypt)

add_executable(lingji-ui-tests tests.cpp theme.cpp)
target_compile_features(lingji-ui-tests PRIVATE cxx_std_17)
target_compile_definitions(lingji-ui-tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A000002)
set_property(TARGET lingji-ui-tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  target_compile_options(lingji-ui-tests PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(lingji-ui-tests PRIVATE -Wall -Wextra -Werror -pthread)
  target_link_options(lingji-ui-tests PRIVATE -static -static-libgcc -static-libstdc++ -municode -pthread)
endif()
target_link_libraries(lingji-ui-tests PRIVATE user32 gdi32 comctl32 bcrypt)
